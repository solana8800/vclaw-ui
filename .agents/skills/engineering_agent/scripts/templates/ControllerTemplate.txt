using System;
using System.Threading;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Mvc;
using MediatR;
using [ROOT_NAMESPACE].Application.Features.[ENTITY_NAME]s.Commands;

namespace [ROOT_NAMESPACE].Api.Controllers
{
    /// <summary>
    /// API Controller tiếp nhận các yêu cầu HTTP bên ngoài liên quan đến dịch vụ [ENTITY_NAME].
    /// </summary>
    [ApiController]
    [Route("api/v1/[ROUTE_NAME]")]
    public class [ENTITY_NAME]sController : ControllerBase
    {
        private readonly IMediator _mediator;

        public [ENTITY_NAME]sController(IMediator mediator)
        {
            _mediator = mediator;
        }

        /// <summary>
        /// Điểm cuối (Endpoint) tiếp nhận yêu cầu POST để khởi tạo một [ENTITY_NAME] mới.
        /// </summary>
        [HttpPost]
        public async Task<IActionResult> Create([FromBody] Create[ENTITY_NAME]Command command, CancellationToken cancellationToken)
        {
            // Điều phối lệnh qua MediatR để xử lý nghiệp vụ ở Application Layer
            var resultId = await _mediator.Send(command, cancellationToken);
            
            // Trả về kết quả 201 Created kèm mã số thực thể vừa tạo
            return CreatedAtAction(nameof(Create), new { id = resultId }, new { Id = resultId });
        }
    }
}
